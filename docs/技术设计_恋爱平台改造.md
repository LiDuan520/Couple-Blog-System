class AlreadyBoundError(BaseAPIException):
    """已绑定"""
    def __init__(self):
        super().__init__(409, "User already bound to a couple", "ALREADY_BOUND")

class InviteNotFoundError(BaseAPIException):
    def __init__(self):
        super().__init__(404, "Invite code not found", "INVITE_NOT_FOUND")

class InviteExpiredError(BaseAPIException):
    def __init__(self):
        super().__init__(410, "Invite code expired", "INVITE_EXPIRED")

class InviteSelfError(BaseAPIException):
    def __init__(self):
        super().__init__(400, "Cannot accept own invite", "INVITE_SELF")

class NotBoundError(BaseAPIException):
    """未绑定（访问 Couple 接口时）"""
    def __init__(self):

## 4. 权限与中间件

### 4.1 `get_current_couple` 依赖

```python
# backend/app/modules/couple/dependencies.py
from fastapi import Depends, HTTPException
from sqlalchemy.orm import Session
from app.core.database import get_db
from app.modules.auth.dependencies import get_current_active_user
from app.modules.couple.repository import CoupleRepository

def get_current_couple(
    current_user: User = Depends(get_current_active_user),
    db: Session = Depends(get_db),
) -> Couple:
    """要求当前用户已绑定"""
    if current_user.couple_id is None:
        raise NotBoundError()
    repo = CoupleRepository(db)
    couple = repo.get_by_id(current_user.couple_id)
    if not couple or not couple.is_active:
        raise NotBoundError()
    return couple
```

### 4.3 邀请码流程（事务）

```python
# backend/app/modules/couple/invite_service.py
def create_invite(db, user):
    if user.couple_id is not None:
        raise AlreadyBoundError()
    # 撤销旧码
    db.query(CoupleInvite).filter_by(created_by_user_id=user.id, used_at=None).update(
        {"expires_at": datetime.utcnow()}  # 立即过期
    )
    code = _generate_code()  # 8 位大写
    invite = CoupleInvite(
        code=code,
        created_by_user_id=user.id,
        expires_at=datetime.utcnow() + timedelta(days=7),
    )
    db.add(invite)
    db.commit()
    return invite

def accept_invite(db, user, code, anniversary_date):
    if user.couple_id is not None:
        raise AlreadyBoundError()
    invite = db.query(CoupleInvite).filter_by(code=code, used_at=None).with_for_update().first()
    if not invite:
        raise InviteNotFoundError()
    if invite.expires_at < datetime.utcnow():
        raise InviteExpiredError()
    if invite.created_by_user_id == user.id:
        raise InviteSelfError()
    if db.query(User).filter_by(id=invite.created_by_user_id, couple_id__ne=None).first():
        raise AlreadyBoundError()  # 邀请人也已绑定

    try:
        # 1. 创建 Couple
        a, b = sorted([invite.created_by_user_id, user.id])
        couple = Couple(user_a_id=a, user_b_id=b, anniversary_date=anniversary_date)
        db.add(couple)
        db.flush()

        # 2. 更新双方 couple_id
        db.query(User).filter(User.id.in_([a, b])).update(
            {"couple_id": couple.id}, synchronize_session=False
        )

        # 3. 标记邀请使用
        invite.used_by_user_id = user.id
        invite.used_at = datetime.utcnow()

        db.commit()
    except IntegrityError:
        db.rollback()
        raise AlreadyBoundError()
    return couple
```

## 5. 时间轴聚合设计

### 5.1 聚合查询（`GET /timeline`）

```python
# backend/app/modules/timeline/service.py
async def get_timeline(
    couple_id: int,
    page: int, page_size: int,
    event_type: Optional[str] = None,
    author_user_id: Optional[int] = None,
    date_from: Optional[date] = None,
    date_to: Optional[date] = None,
    order: str = "desc",
):
    # 1. 收集原始事件
    blogs = await blog_repo.get_timeline_blogs(couple_id, ...)
    anniversaries = ann_repo.get_timeline_anniversaries(couple_id, ...)
    photos = photo_repo.get_timeline_photos(couple_id, ...)
    milestones = compute_milestones(couple.anniversary_date, date_from, date_to)

    # 2. 合并排序
    events = []
    events.extend(map_blog_to_event(blogs))
    events.extend(map_anniversary_to_event(anniversaries))
    events.extend(map_photo_to_event(photos))
    events.extend(milestones)
    events.sort(key=lambda e: e.occurred_at, reverse=(order == "desc"))

    # 3. 分页
    total = len(events)
    start = (page - 1) * page_size
    page_items = events[start:start + page_size]

    return {
        "items": page_items,
        "total": total,
        "page": page, "page_size": page_size, "total_pages": ceil(total / page_size),
        "milestones": compute_milestones(couple.anniversary_date, ...),
        "next_countdown": compute_next_countdown(couple, anniversaries, limit=3),
    }
```

### 5.2 里程碑计算（系统事件）

```python
MILESTONE_DAYS = [100, 200, 365, 500, 730, 1000, 1500, 2000, 3000, 5000]

def compute_milestones(anniversary_date, date_from, date_to):
    today = date.today()
    days_together = (today - anniversary_date).days
    events = []
    for d in MILESTONE_DAYS:
        occurred = anniversary_date + timedelta(days=d)
        if date_from and occurred < date_from: continue
        if date_to and occurred > date_to: continue
        events.append({
            "id": f"milestone-{d}",
            "type": "MILESTONE",
            "title": f"在一起 {d} 天",
            "summary": _milestone_emoji(d),
            "occurred_at": datetime(occurred.year, occurred.month, occurred.day),
            "cover_url": None,
            "author": None,
            "payload": {"days": d, "is_future": occurred > today},
        })
    return events
```

### 5.3 性能考虑

- 三个独立查询（blog/anniversary/photo），每个查询在 PG/Mongo 上都很快
- 合并 + 排序发生在 Python，**用 list.sort 不依赖 DB 排序**
- 缓存键：`cache:timeline:couple:<id>:<query_hash>`，TTL 60s
- 写操作 invalidate：`/blogs/*`、`/anniversaries/*`、`/photos/*` 写后 `SCAN cache:timeline:couple:<id>:*` 删除
- 量化：couple 一年平均 200 事件，合并排序 < 5ms

### 6.2 缓存失效的统一接口

# backend/app/core/cache.py（新建）
def invalidate_couple_caches(couple_id: int, patterns: list[str] = None):
    r = get_redis()
    if patterns is None:
        patterns = [
            f"cache:blog:list:couple:{couple_id}:*",
            f"cache:timeline:couple:{couple_id}:*",
            f"cache:dashboard:couple:{couple_id}",
        ]
    for pattern in patterns:
        for key in r.scan_iter(pattern):
            r.delete(key)

## 8. 数据迁移

### 8.1 v1 → v2 迁移脚本

```python
# backend/scripts/migrate_v1_to_v2.py
"""
执行步骤：
1. 给 users 表加 couple_id 列
2. 创建 couples / couple_invites / anniversaries / albums / photos 表
3. 给 MongoDB blogs 加 couple_id 字段（v1 数据 couple_id=null）
4. 不动老数据，允许老用户继续以"未绑定"模式访问博客
"""
import asyncio
from sqlalchemy import text
from app.core.database import engine, mongodb_db

async def migrate():
    # 1. PG 加列
    with engine.begin() as conn:
        conn.execute(text("ALTER TABLE users ADD COLUMN IF NOT EXISTS couple_id INT REFERENCES couples(id);"))
        conn.execute(text("CREATE UNIQUE INDEX IF NOT EXISTS idx_users_couple_unique ON users(couple_id) WHERE couple_id IS NOT NULL;"))

    # 2. 创建新表（通过 Base.metadata.create_all）
    from app.core.database import Base
    from app.modules.couple.models import Couple, CoupleInvite
    from app.modules.anniversary.models import Anniversary
    from app.modules.album.models import Album, Photo
    Base.metadata.create_all(engine, tables=[Couple.__table__, CoupleInvite.__table__,
                                              Anniversary.__table__, Album.__table__, Photo.__table__])

    # 3. MongoDB 加索引
    await mongodb_db.blogs.create_index([("couple_id", 1), ("is_deleted", 1), ("event_date", -1)])

    print("Migration completed.")

if __name__ == "__main__":
    asyncio.run(migrate())
```

## 11. 风险与缓解

| v1 blog 数据迁移 | 低 | 脚本 dry-run 模式；保留回滚；**v2 范围内不动老数据**，老博客 `couple_id=null` |
| 情侣解绑逻辑复杂 | 低 | v2 范围外，留 DELETE 接口占位；软删不撤销 |
| 照片 EXIF 隐私 | 低 | v2 不解析 EXIF；只存 taken_at（用户填）|
| 双人协作 UI 冲突 | 中 | 同一资源同时编辑：last-write-wins（v2 简单方案） |
| 软删数据无限增长 | 中 | 见下方"软删回收 cron 任务" |
| 邀请码被暴力枚举 | 中 | 8 位 + 32^8 ≈ 1.1 万亿组合；加 rate limit 5/min；记录异常并告警 |

### 11.1 软删回收 cron 任务

软删 30 天后真删由以下机制实现（v2 范围内）：

1. 在 `app/modules/anniversary/service.py` 等位置实现 `purge_soft_deleted(days: int = 30)` 函数
2. 在 `app/main.py` lifespan `startup` 中注册 APScheduler 定时任务（**每日凌晨 3 点**执行）
3. 引入依赖：`apscheduler==3.10.*`（追加到 requirements.txt）
4. 仅真删 `is_deleted=true AND updated_at < now() - 30d` 的记录

> 替代方案：v2 范围内不实现 cron，靠 DB 容量评估 + 手动 SQL 删除；v2.1 再上 APScheduler。本期任务**不在 M-15 验收范围**。

### 12.1 邀请码生成

```python
import secrets
import string

def _generate_invite_code(length: int = 8) -> str:
    """生成 8 位大写字母+数字的邀请码（去掉易混的 0/O/1/I/L）"""
    alphabet = "".join(c for c in (string.ascii_uppercase + string.digits) if c not in "0O1IL")
    return "".join(secrets.choice(alphabet) for _ in range(length))
```

### 12.2 里程碑事件生成

```python
from datetime import date, datetime, timedelta

MILESTONE_DAYS = [100, 200, 365, 500, 730, 1000, 1500, 2000, 3000, 5000]

def compute_milestones(anniversary_date: date, limit: int = 3) -> list[dict]:
    today = date.today()
    events = []
    for d in MILESTONE_DAYS:
        occurred = anniversary_date + timedelta(days=d)
        events.append({
            "days": d,
            "occurred_at": datetime.combine(occurred, datetime.min.time()),
            "is_future": occurred > today,
        })
    # 最近的 3 个（含过去 + 未来）
    past = sorted([e for e in events if not e["is_future"]], key=lambda e: -e["days"])[:1]
    future = sorted([e for e in events if e["is_future"]], key=lambda e: e["days"])[:2]
    return past + future
```
